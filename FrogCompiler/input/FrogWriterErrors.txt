/* Scenario 1 (active): every token and every production in this file is
   perfectly valid Frog - the Scanner and Parser accept it with zero errors.
   The problem only shows up at runtime, when the Writer actually evaluates
   the expression: dividing by a variable that holds zero. This proves the
   Writer's own semantic checks (not just syntax checks) are working, and
   that it reports the failure with a line number and stops instead of
   crashing silently. */

	tadpole a;
	tadpole b;
	a = 10;
	b = 0;
	print(a);
	a = a / b;
	print(a);
	leap 0;
}
